先聚合，再压缩

设计：交接文档 §五先写压缩，再写模式聚合；早期设计 §3.4 还写了完成后立即 summarized。

complete_task 先生成并保存摘要与难度，再聚合，最后扫描压缩。完成不等于删除原文；保护规则和冷却期可以让任务继续保持 raw。这样可先保护刚形成重复模式的早期证据。

 468      def complete_task(self, task_id: str, auto_maintain: bool | None = None) -> CompletionResult:
 469          """标记完成，并按交接文档第五节把整个循环走完。
 470  
 471          文档规定的 `[Session完成]` 分支是三步：
 472  
 473              status → 已完成
 474              → 生成 summary，触发压缩流程（Demote/Expire 规则判断）
 475              → 检测跨 session 重复模式 → 更新语义记忆（Promote 规则判断）
 476  
 477          这里三步都做。**顺序是先聚合再压缩**，和文档写的顺序相反，是有意的：
 478          刚完成的这条 session 可能正是让某个模式成立的第二份证据，而那个模式
 479          要保护的是**更早的那条**——它此刻可能刚好冷却到期该被压缩了。
 480          先压后聚的话，证据会在模式成立前一刻被压掉。
 481  
 482          摘要仍然在这里生成（而不是留给压缩阶段）：续接提示和检索都要用它，
 483          提前生成还能让后续 SUMMARIZE 分支直接复用，不必再调一次 LLM。
 484          """
 485          task = self.store.get_task(task_id)
 486          if task is None:
 487              raise KeyError(f"找不到 task: {task_id}")
 488          task.status = STATUS_DONE
 489          task.completed_at = self.now_iso()
 490          task.summary = self.summarize(task)
 491          # 趁对话历史还在，把困难程度固化下来——压缩之后就算不出来了
 492          task.difficulty_score = compute_difficulty(task)
 493          self._refresh_memory_fields(task)
 494          self.store.update_task(task)
 495  
 496          result = CompletionResult(task=task)
 497          if auto_maintain is None:
 498              auto_maintain = auto_maintain_enabled()
 499          if not auto_maintain:
 500              return result
 501  
 502          result.consolidation = self.consolidate_memory()
 503          result.maintenance = self.run_maintenance()
 504          # 维护可能把这条 session 自己也改了，回读最新状态
 505          refreshed = self.store.get_task(task_id)
 506          if refreshed is not None:
 507              result.task = refreshed
 508          return result

 202  def consolidate(
 203      tasks: Sequence[TaskSession],
 204      existing: Sequence[SemanticMemoryEntry],
 205      *,
 206      embed: Any,
 207      name_pattern: Any = None,
 208      min_occurrences: int = DEFAULT_MIN_OCCURRENCES,
 209      min_difficulty: float = DEFAULT_MIN_DIFFICULTY,
 210      cluster_threshold: float = DEFAULT_CLUSTER_THRESHOLD,
 211      merge_threshold: float = DEFAULT_MERGE_THRESHOLD,
 212      lock_occurrences: int = DEFAULT_LOCK_OCCURRENCES,
 213  ) -> ConsolidationReport:
 214      """把已完成 session 聚合成语义记忆条目。
 215  
 216      `embed` 是 `(list[str]) -> list[list[float]]`；
 217      `name_pattern` 是可选的 `(Cluster) -> str`，不给就用规则命名。
 218      纯函数式：不碰存储，返回要写入的条目，由调用方持久化。
 219      """
 220      # 只有真的卡过的 session 才够格当"重复困难模式"的证据。
 221      # 没有这道门槛，反复做的杂事会被当成困难，白白锁住不让压缩。
 222      done = sorted(
 223          (
 224              t
 225              for t in tasks
 226              if t.status == STATUS_DONE and t.difficulty_score >= min_difficulty
 227          ),
 228          key=lambda t: t.created_at,
 229      )
 230      all_done = sum(1 for t in tasks if t.status == STATUS_DONE)
 231      report = ConsolidationReport(
 232          considered_sessions=len(done),
 233          filtered_easy=all_done - len(done),
 234      )
 235      if not done:
 236          return report
 237  
 238      texts = [difficulty_text(t) for t in done]
 239      embeddings = embed(texts)
 240      clusters = cluster_sessions(done, embeddings, threshold=cluster_threshold)
 241  
 242      existing_list = list(existing)
 243      for cluster in clusters:
 244          if len(cluster.members) < min_occurrences:
 245              report.skipped_clusters += 1
 246              continue
 247  
 248          evidence = [t.id for t in cluster.members]
 249          occurrences = len(evidence)
 250  
 251          # 和已有条目重合的话走更新，避免每次跑都新建一堆重复模式
 252          match: SemanticMemoryEntry | None = None
 253          best_sim = merge_threshold
 254          for entry in existing_list:
 255              sim = cosine_similarity(cluster.centroid, entry.embedding)
 256              if sim >= best_sim:
 257                  match, best_sim = entry, sim
 258  
 259          if match is None:
 260              name = (name_pattern(cluster) if name_pattern else None) or rule_based_pattern_name(cluster)
 261              entry = SemanticMemoryEntry.create(name)
 262              entry.evidence_session_ids = evidence
 263              entry.occurrence_count = occurrences
 264              entry.confidence = confidence_for(occurrences)
 265              entry.last_seen = now_iso()
 266              entry.embedding = list(cluster.centroid)
 267              entry.locked = occurrences >= lock_occurrences
 268              report.created.append(entry)
 269              existing_list.append(entry)
 270          else:
 271              # 证据只增不减：已有的证据即使这次没进簇也保留，
 272              # 否则会话被归档后模式会莫名其妙地"退化"
 273              merged = list(dict.fromkeys(match.evidence_session_ids + evidence))
 274              match.evidence_session_ids = merged
 275              match.occurrence_count = len(merged)
 276              match.confidence = confidence_for(match.occurrence_count)
 277              match.last_seen = now_iso()
 278              match.embedding = list(cluster.centroid)
 279              match.locked = match.occurrence_count >= lock_occurrences
 280              report.updated.append(match)
 281  
 282      return report

 285  def matching_pattern(
 286      task: TaskSession,
 287      entries: Sequence[SemanticMemoryEntry],
 288      query_embedding: Sequence[float] | None,
 289      *,
 290      min_similarity: float = 0.35,
 291      min_occurrences: int = DEFAULT_MIN_OCCURRENCES,
 292  ) -> SemanticMemoryEntry | None:
 293      """当前 session 命中了哪条重复困难模式。
 294  
 295      自己就是某条模式的证据时直接命中，不必再比相似度——
 296      这条捷径让刚被聚合进去的 session 立刻就能享受到巩固保留。
 297      """
 298      for entry in entries:
 299          if task.id in entry.evidence_session_ids and entry.occurrence_count >= min_occurrences:
 300              return entry
 301  
 302      best: SemanticMemoryEntry | None = None
 303      best_sim = min_similarity
 304      for entry in entries:
 305          if entry.occurrence_count < min_occurrences:

 307          sim = cosine_similarity(query_embedding, entry.embedding)
 308          if sim >= best_sim:
 309              best, best_sim = entry, sim
 310      return best

 122          self, text: str, day: str | None = None, auto_maintain: bool | None = None
 123      ) -> tuple[DaySession, list[TaskSession]]:
 124          """排一天的任务。顺带跑一次记忆维护。
 125  
 126          维护为什么要挂在这里：归档（Expire）是**时间驱动**的——闲置够久才降级。
 127          只在 complete_task 里跑维护的话，一段时间没完成任何任务就没有触发点，

 129          """

 131          if auto_maintain is None:

 139              task = TaskSession.create(day_id=day, title=title)
 140              self._refresh_memory_fields(task)
 141              self.store.create_task(task)
 142              tasks.append(task)
